cmake_minimum_required(VERSION 3.16)
if(TARGET xr)
  target_include_directories(xr PUBLIC ${CMAKE_CURRENT_LIST_DIR})
else()
  project(TalosLink LANGUAGES CXX)
  add_library(TalosLink INTERFACE)
  target_include_directories(TalosLink INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
  target_compile_features(TalosLink INTERFACE cxx_std_20)
  include(CTest)
  if(BUILD_TESTING)
    add_executable(talos_link_test tests/talos_link_test.cpp)
    target_link_libraries(talos_link_test PRIVATE TalosLink)
    target_compile_options(talos_link_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    add_test(NAME talos_link_test COMMAND talos_link_test)
  endif()
endif()
